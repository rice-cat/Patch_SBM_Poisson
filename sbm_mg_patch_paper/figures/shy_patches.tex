    \newcommand{\drawCubicDoFsAt}[2]{%

    \newcommand{\drawShySquareAt}[2]{%
        \fill[gray] ($(#1,#2)+(-0.1,-0.1)$) rectangle ($(#1,#2)+(0.1,0.1)$);%
    }
